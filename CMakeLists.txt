cmake_minimum_required(VERSION 3.18)
project(parallel_image_denoising LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_OPENMP "Build OpenMP support" ON)
option(ENABLE_MPI "Build MPI executable" ON)
option(ENABLE_CUDA "Build CUDA and hybrid executables" OFF)
include(CTest)

add_library(denoise_core
    src/image.c
    src/noise.c
    src/metrics.c
    src/timing.c
    src/serial/bilateral_serial.c
)
target_include_directories(denoise_core PUBLIC include)
if(NOT MSVC)
    target_link_libraries(denoise_core PUBLIC m)
endif()

set(HAVE_OPENMP 0)
if(ENABLE_OPENMP)
    find_package(OpenMP COMPONENTS C QUIET)
    if(OpenMP_C_FOUND)
        set(HAVE_OPENMP 1)
        target_sources(denoise_core PRIVATE src/openmp/bilateral_openmp.c)
        target_link_libraries(denoise_core PUBLIC OpenMP::OpenMP_C)
        message(STATUS "OpenMP C enabled")
    else()
        target_sources(denoise_core PRIVATE src/openmp/bilateral_openmp_stub.c)
        message(WARNING "OpenMP C unavailable; OpenMP mode disabled")
    endif()
else()
    target_sources(denoise_core PRIVATE src/openmp/bilateral_openmp_stub.c)
endif()
target_compile_definitions(denoise_core PUBLIC HAVE_OPENMP=${HAVE_OPENMP})

add_executable(denoise src/main.c src/cuda/cuda_stubs.c)
target_link_libraries(denoise PRIVATE denoise_core)

set(HAVE_MPI 0)
if(ENABLE_MPI)
    find_package(MPI COMPONENTS C QUIET)
    if(MPI_C_FOUND)
        set(HAVE_MPI 1)
        add_executable(denoise_mpi src/mpi/bilateral_mpi.c)
        target_link_libraries(denoise_mpi PRIVATE denoise_core MPI::MPI_C)
        message(STATUS "MPI C enabled")
    else()
        message(STATUS "MPI C unavailable; MPI target skipped")
    endif()
endif()

if(ENABLE_CUDA)
    include(CheckLanguage)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
        enable_language(CUDA)
        add_library(denoise_gpu src/cuda/bilateral_cuda.cu src/hybrid/bilateral_hybrid.cu)
        target_include_directories(denoise_gpu PUBLIC include)
        target_link_libraries(denoise_gpu PUBLIC denoise_core)
        set_target_properties(denoise_gpu PROPERTIES CUDA_STANDARD 11 CUDA_STANDARD_REQUIRED ON)
        add_executable(denoise_cuda src/main.c)
        target_compile_definitions(denoise_cuda PRIVATE HAVE_CUDA=1)
        target_link_libraries(denoise_cuda PRIVATE denoise_core denoise_gpu)
    else()
        message(WARNING "CUDA requested but nvcc unavailable; CUDA targets skipped")
    endif()
endif()

if(BUILD_TESTING)
    add_executable(test_core tests/c/test_core.c)
    target_link_libraries(test_core PRIVATE denoise_core)
    add_test(NAME core COMMAND test_core)
endif()

configure_file(include/build_config.h.in generated/build_config.h @ONLY)
target_include_directories(denoise PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(TARGET denoise_cuda)
    target_include_directories(denoise_cuda PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
endif()
message(
    STATUS "Capabilities: OpenMP=${HAVE_OPENMP}, MPI=${HAVE_MPI}, CUDA requested=${ENABLE_CUDA}")
